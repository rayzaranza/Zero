    PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}/src"
)

    PUBLIC
        "${CMAKE_CURRENT_SOURCE_DIR}/${ZERO_SRC}/zrpch.h"
)

target_link_libraries(Zero
    PUBLIC
	    glm::glm 
	    glad::glad 
	    glfw
	    spdlog::spdlog
        imgui::imgui
        ${Stb_INCLUDE_DIR}
)

target_compile_definitions(Zero
    PUBLIC
        ZERO_PLATFORM_WINDOWS
)

target_compile_definitions(Zero
    PUBLIC
        ZERO_ENABLE_ASSERTS
        GLFW_INCLUDE_NONE
        ZERO_BUILD_DLL
        IMGUI_IMPL_OPENGL_LOADER_GLAD
)
